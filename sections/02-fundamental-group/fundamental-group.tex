\subsubsection{The Group Structure on the Fundamental Group}

\begin{definition}{Fundamental Group Functor}{}
  Define the functor \[
  \pi_1\coloneq [(S^1,e_1),\cdot]_*:\hToppt \to \Set
  \]
\end{definition}

We aim to understand this functor geometrically. Take $(X,x_0)\in\Toppt$. Then \[
\pi_1(X,x_0)=[(S^1,e_1),(X,x_0)]_* = \Toppt((S^1,e_1),(X,x_0))/\simeq_*
\]
is precisely the set of pointed homotopy classes of pointed loops in $(X,x_0)$.

\begin{remark}{Parametrisation of $S^1$}{}
  We frequently use the homeomorphism $[0,1]/(0\sim 1)\to S^1, [t]\mapsto e^{2\pi i t}$ allowing us to parametrise points on $S^1$ by elements of $[0,1]$.
\end{remark}

We can concatenate loops at the same basepoint to form a new loop, and this constitutes the group structure on $\pi_1$.

\begin{theorem}{$\pi_1(X,x_0)\in\Grp$, the Fundamental Group}{fgg}
  Let $(X,x_0)\in\Toppt$. \begin{enumerate}
    \item The following composition map is well-defined \begin{align*}
      \circ : \pi_1(X,x_0)\times\pi_1(X,x_0)&\to \pi_1(X,x_0)\\
      ([\gamma]_*,[\eta]_*) &\mapsto \left[
        \gamma\ast\eta \coloneq [t]\mapsto \begin{cases}
          \gamma([2t]) & t\in[0,0.5]\\
          \eta([2t-1]) & t\in[0.5,1]
        \end{cases}
      \right]
    \end{align*}
    \item $\pi_1(X,x_0)$ forms a group with this concatenation. This is the \textbf{fundamental group} of $(X,x_0)$.
    \item Let $(f:(X,x_0)\to(Y,y_0))\in\Toppt$. Then $\pi_1(f)$ as defined below is a group homomorphism. \begin{align*}
           \pi_1(f)=\Toppt((S^1,1),f):\pi_1(X,x_0)&\to \pi_1(Y,y_0) \\
           [\gamma]_*&\mapsto[f\circ\gamma]_*
    \end{align*}
  \end{enumerate}
\end{theorem}

\begin{proof}[\proofof{th:fgg}]
  \paragraph{Composition is Well Defined.} Obviously if $\gamma,\eta\in\Toppt((S^1,e_1),(X,x_0))$ then $\gamma\ast\eta$ is a well-defined continuous loop based at $x_0$. Next, if $\gamma\simeq_* \gamma'$ (via $h$) and $\eta\simeq_* \eta'$ (via $k$) then we can define \begin{align*}
    (S^1,e_1)\times [0,1] &\to (X,x_0) \\
    ([s],t) &\mapsto \begin{cases}
      h([2s],t) & s\in[0,0.5]\\
      k([2s-1],t) & s\in[0.5,1]
    \end{cases}
  \end{align*}
  and see that this is a pointed homotopy equivalence, so $\gamma\ast\eta\simeq_*\gamma'\ast\eta'$.\\

  \paragraph{Satisfies Group Axioms.} \begin{itemize}
    \item \textbf{Neutral Element.} This is the constant loop $c:(S^1,e_1)\to(X,x_0)$. Indeed, $[c]_*\in \pi_1(X,x_0)$ is neutral wrt. $\circ$. Explicitly, for every $\gamma\in\Toppt((S^1,e_1),(X,x_0))$ we construct the pointed homotopy equivalence \begin{align*}
      (S^1,e_1)\times[0,1] &\to (X,x_0) \\
      ([s],t) &\mapsto \begin{cases}
        \gamma([2s(1+t)\inv]) & s\in[0,0.5(1+t)] \\
        x_0 & s\in[0.5(1+t),t]
      \end{cases}
    \end{align*}
    to show that $c\ast \gamma\simeq_* \gamma$. A similar argument shows that $\gamma\ast c\simeq_* \gamma$.
    \item \textbf{Inverse Elements.} Let $\gamma\in\Toppt((S^1,e_1),(X,x_0))$. We walk the loop backwards: \begin{align*}
      \overline{\gamma} : (S^1,e_1) &\to (X,x_0) \\
      [s] &\mapsto \gamma([1-s])
    \end{align*}
    and we obtain $\gamma\ast\overline{\gamma}\simeq_*c$ through the following pointed homotopy equivalence: \begin{align*}
      (S^1,e_1)\times[0,1] &\to (X,x_0)\\
      ([s],t)&\mapsto \begin{cases}
        x_0 & s\in[0,0.5t]\cup[1-0.5t,1]\\
        \gamma([2s-t]) & s\in[0.5t,0.5]\\
        \overline{\gamma}([2s-1+t]) & s\in[0.5,1-0.5t]
      \end{cases}
    \end{align*}
    And similarly $\overline{\gamma}\ast\gamma\simeq_* c$. So $[\overline{\gamma}]_*$ is the two-sided inverse of $[\gamma]_*$.
    \item Let $\alpha,\beta,\gamma\in\Toppt((S^1,e_1),(X,x_0))$. We can form a homotopy as in Figure \ref{fig:grpaxioms}(c) to show that \[(\alpha\ast\beta)\ast\gamma \simeq_* \alpha\ast(\beta\ast\gamma)\]
  \end{itemize}

  \paragraph{Pointed Homotopies go to Homomorphisms} Let $\gamma,\eta\in\Toppt((S^1,e_1),(X,x_0))$. By definition we can see that $f\circ(\gamma\ast\eta)=(f\circ\gamma)\ast(f\circ\eta)$. In particular we have \[
    \pi_1([f]_*)([\gamma]_* \cdot [\eta]_*) =  \pi_1([f]_*)([\gamma]_*) \cdot  \pi_1([f]_*)([\eta]_*)
  \]
\end{proof}

\begin{remark}{$\pi_1$ as a Group-Valued Functor}{pi1grp}
  The group structure is compatible with the maps $\pi_1(f)$ and with identities, so $\pi_1$ lifts along the forgetful functor $\Grp\to\Set$: \[
  \begin{tikzcd}
    & \Grp \ar[d, "\text{forget}"] \\
    \hToppt \ar[ur, dashed, "\pi_1"] \ar[r, "\pi_1"'] & \Set
  \end{tikzcd}
  \]
  We also write $\pi_1$ for the lift $\hToppt\to\Grp$ and for its composite with $\Toppt\to\hToppt$.
\end{remark}

The proof of Theorem \ref{th:fgg} only used structure on the pointed circle: $(S^1,e_1)$ is a \emph{cogroup object} in $\hToppt$, which is what makes $[(S^1,e_1),\cdot]_*$ group-valued. The unpointed circle has no such structure in $\hTop$, so $[S^1,\cdot]:\hTop\to\Set$ does not lift to $\Grp$; basepoints are essential, not an artefact of the construction.\\

The same idea works in higher dimensions. For $n\geq 2$, identify $S^n$ with $[0,1]^n/\partial[0,1]^n$, so that elements of $\pi_n(X,x_0)\coloneq[(S^n,e_1),(X,x_0)]_*$ are represented by maps $[0,1]^n\to X$ sending $\partial[0,1]^n$ to $x_0$. Concatenating in the $j$-th coordinate (running $\gamma$ for $t_j\in[0,\tfrac12]$ and $\eta$ for $t_j\in[\tfrac12,1]$, rescaled) gives a group structure $+_j$ on $\pi_n(X,x_0)$ by the same arguments as for $\pi_1$. Since there is now a second coordinate in which to slide $\gamma$ and $\eta$ past each other, all the $+_j$ coincide and are commutative (Eckmann--Hilton): the \emph{higher homotopy groups} $\pi_n(X,x_0)$ are abelian.

\begin{example}{Trivial Fundamental Groups}{pi1triv}
  \begin{itemize}
    \item If $(X,x_0)$ is pointedly contractible, then $\pi_1(X,x_0)$ is trivial, since $(X,x_0)\cong(\pt,\pt)$ in $\hToppt$ and $\pi_1$ is a functor. Merely contractible spaces also have trivial $\pi_1$, but this needs care: a free null-homotopy of a loop $\gamma$ may drag the basepoint along a path $\eta$ (Figure \ref{fig:pi1triv}(a)), and one must still show $\gamma\simeq_*\eta\ast\bar\eta\simeq_* c$ (Exercise; cf.\ change of basepoint).
    \item If $n\geq 2$, then $\pi_1(S^n,e_1)$ is trivial. The tempting argument is that $\gamma$ misses some point $p$, and $S^n\setminus\{p\}\cong\R^n$ is contractible. But a loop may be \emph{surjective} onto $S^n$ (Figure \ref{fig:pi1triv}(b)). The fix: by uniform continuity, subdivide $[0,1]$ so that each piece of $\gamma$ lies in an open hemisphere, and straighten each piece to a great-circle arc via the normalised straight-line homotopy, fixing its endpoints. The result is pointedly homotopic to $\gamma$, consists of finitely many arcs, and so misses some $p$ (Figure \ref{fig:pi1triv}(c)). Alternatively, use Seifert--van Kampen.
  \end{itemize}
\end{example}

\subsubsection{Changing the Basepoint \& the Fundamental Groupoid}

$\pi_1(X,x_0)$ sees only the path-connected component where $x_0$ lies. It is easy to see that basepoints in the same path-connected component lead to isomorphic fundamental groups. However, no \textbf{canonical} isomorphism generally exists.

\begin{theorem}{Change of Basepoint in Fundamental Groups}{pi1cob}
  Let $X\in\Top$ and $x,y\in X$. Let $\eta,\eta':[0,1]\to X$ be paths in $X$ from $x$ to $y$. Then \begin{enumerate}
    \item The following is a well-defined group isomorphism: \begin{align*}
      \phi_\eta:\pi_1(X,y)&\to\pi_1(X,x) \\
                [\gamma]_*&\mapsto[(\eta\ast\gamma)\ast\overline{\eta}]_*
    \end{align*}
    \item We have $c_h\circ\phi_\eta=\phi_{\eta'}$ where \begin{align*}
      c_h:\pi_1(X,x)&\to\pi_1(X,x)\\ g&\mapsto hgh\inv
    \end{align*}
    is conjugation by $h=[\eta'\ast\overline{\eta}]_*\in\pi_1(X,x)$.
  \end{enumerate}
\end{theorem}

\begin{proof}[\proofof{th:pi1cob}]
  \paragraph{Claim 1.} $\phi_\eta$ is easily shown to be well-defined and compatible with the group structures of $\pi_1(X,\cdot)$. We also see that $\phi_{\overline\eta}$ is the inverse of $\phi_\eta$, so $\phi_\eta$ is a group isomorphism.
  \paragraph{Claim 2.} By construction $\eta'\ast\overline\eta$ is a loop based at $x$. Thus $h=[\eta'\ast\overline\eta]_*$ is indeed an element of $\pi_1(X,x)$. Moreover for $\gamma\in\Toppt((S^1,e_1),(X,y))$ we compute \begin{align*}
    \phi_{\eta'}([\gamma]_*) &= [\eta'\ast\gamma\ast\overline{\eta'}]_*\\
    &= [\eta'\ast(\overline\eta\ast\eta)\ast\gamma\ast(\overline\eta\ast\eta)\ast\overline{\eta'}]_* \\
    &= h\cdot[\eta\ast\gamma\ast\overline\eta]_*\cdot h\inv \\
    &= c_h\circ \phi_\eta([\gamma]_*)
  \end{align*}
\end{proof}

Fixing a basepoint can be inconvenient. In such cases we may consider the fundamental groupoid, which keeps all basepoints at once.

\begin{definition}{Fundamental Groupoid}{fgpd}
  Let $X\in\Top$. For $x,y\in X$ let $P_X(x,y)\coloneq\{\eta\in\Top([0,1],X) : \eta(0)=x,\ \eta(1)=y\}$ be the set of paths from $x$ to $y$, and write $\eta\simeq_\partial\eta'$ if $\eta\simeq\eta'$ \emph{rel $\{0,1\}$}, i.e.\ via a homotopy $h$ with $h(0,t)=x$ and $h(1,t)=y$ for all $t\in[0,1]$. The \textbf{fundamental groupoid} $\Pi_1(X)$ is the category with
  \begin{itemize}
    \item objects $\Ob(\Pi_1(X))\coloneq X$, the underlying set;
    \item morphisms $\Pi_1(X)(x,y)\coloneq P_X(x,y)/\simeq_\partial$;
    \item composition $[\eta']\circ[\eta]\coloneq[\eta\ast\eta']$ for $[\eta]\in\Pi_1(X)(x,y)$ and $[\eta']\in\Pi_1(X)(y,z)$;
    \item identities $\id_x\coloneq[c_x]$, the class of the constant path at $x$.
  \end{itemize}
\end{definition}

\begin{proposition}{$\Pi_1$ is a Functor $\Top\to\Grpd$}{fgpdfun}
  \begin{enumerate}
    \item $\Pi_1(X)$ is a groupoid, with $[\eta]\inv=[\overline\eta]$.
    \item Every $f\in\Top(X,Y)$ induces a functor $\Pi_1(f):\Pi_1(X)\to\Pi_1(Y)$, $x\mapsto f(x)$, $[\eta]\mapsto[f\circ\eta]$, and this makes $\Pi_1:\Top\to\Grpd$ a functor.
    \item A homotopy $h:f\simeq g$ induces a natural isomorphism $\Pi_1(f)\Rightarrow\Pi_1(g)$ with components $[h(x,\cdot)]\in\Pi_1(Y)(f(x),g(x))$. In particular, homotopy equivalences induce equivalences of groupoids.
  \end{enumerate}
\end{proposition}

\begin{proof}[\proofof{th:fgpdfun}]
  \paragraph{Claim 1.} The homotopies in the proof of Theorem \ref{th:fgg} (Figure \ref{fig:grpaxioms}) fix $s=0$ and $s=1$ throughout and never use that the endpoints agree, so they are homotopies rel $\{0,1\}$ between paths. Hence composition is well defined, associative and unital, and $[\overline\eta]$ is inverse to $[\eta]$.
  \paragraph{Claim 2.} $f\circ(\eta\ast\eta')=(f\circ\eta)\ast(f\circ\eta')$, $f\circ c_x=c_{f(x)}$, and $f$ carries homotopies rel $\{0,1\}$ to homotopies rel $\{0,1\}$. Functoriality in $f$ is clear.
  \paragraph{Claim 3.} Write $h_x\coloneq h(x,\cdot)$. Naturality at $[\eta]\in\Pi_1(X)(x,y)$ asks that $(f\circ\eta)\ast h_y\simeq_\partial h_x\ast(g\circ\eta)$. Both are images under $H:[0,1]^2\to Y$, $(s,t)\mapsto h(\eta(s),t)$, of boundary paths of the square from $(0,0)$ to $(1,1)$, and these are homotopic rel endpoints in the convex set $[0,1]^2$. If $g\circ f\simeq\id_X$ and $f\circ g\simeq\id_Y$, then $\Pi_1(g)\circ\Pi_1(f)\cong\id$ and $\Pi_1(f)\circ\Pi_1(g)\cong\id$, so $\Pi_1(f)$ is an equivalence.
\end{proof}

\begin{remark}{$\Pi_1$ versus $\pi_1$}{fgpdpi1}
  \begin{itemize}
    \item \textbf{Automorphisms.} Via $S^1\cong[0,1]/\{0,1\}$, loops at $x$ are paths from $x$ to $x$, so $\Aut_{\Pi_1(X)}(x)=\pi_1(X,x)^{\mathrm{op}}$; the ``op'' is only because $\circ$ is written right-to-left while $\ast$ is left-to-right, and $g\mapsto g\inv$ identifies it with $\pi_1(X,x)$.
    \item \textbf{Objects.} Isomorphism classes of objects of $\Pi_1(X)$ are exactly the path components of $X$.
    \item \textbf{Change of basepoint.} A path $\eta$ from $x$ to $y$ is an isomorphism $[\eta]:x\to y$ in $\Pi_1(X)$, and Theorem \ref{th:pi1cob} is conjugation by it. Different choices of $\eta$ differ by an automorphism of $x$, hence the non-canonical isomorphism.
    \item \textbf{No basepoint needed.} If $X$ is path-connected, the inclusion of the full subcategory on $\{x\}$ is an equivalence of groupoids, so $\Pi_1(X)\simeq\pi_1(X,x)$ (as a one-object groupoid) for \emph{every} $x$; $\Pi_1(X)$ is this group with the choice of basepoint removed.
    \item \textbf{Homotopy invariance.} By Claim 3, homotopic maps give \emph{naturally isomorphic}, not equal, functors. So $\Pi_1$ does not factor through $\hTop$ to $\Grpd$; it factors to the category of groupoids and isomorphism classes of functors.
  \end{itemize}
\end{remark}

The fundamental groupoid makes many statements about $\pi_1$ cleaner at the cost of working with groupoids instead of groups. One can go further: adding homotopies between paths as $2$-morphisms, homotopies between those as $3$-morphisms, and so on, gives the \emph{fundamental $\infty$-groupoid} of $X$. This is the starting point for modern homotopy theory: $\infty$-categories, homotopy type theory (HoTT) and anima.

\bigskip
We can work with the basepoints to define notions of connectedness.

\begin{definition}{Simply Connected}{}
  A non-empty $X\in\Top$ is simply connected if it is path-connected and $\exists x_0\in X:\pi_1(X,x_0)\iso\{0\}$.\\
  Note that if one such $x_0$ works then every basepoint will.
\end{definition}

\begin{definition}{Higher Connectedness}{}
  Let $n\in\N$. Then $X\in\Top$ is $n$-connected if \[
    \forall_{j\in0..n} \ \forall_{x_0\in X} \ \#\pi_n(X,x_0)=1
  \]
\end{definition}

In this definition the condition on $\pi_0$ ensures that $X$ is at least path-connected, so $0$-connected means path-connected and $1$-connected means simply connected.

% The banner is full-bleed left/right, and also fills the top margin (sections start a new page).
\newcommand{\sectionbanner}[2]{%
  {\setlength{\fboxsep}{0pt}%
  \noindent\makebox[\linewidth][l]{%
  \tikz[remember picture,overlay]{\coordinate (sbpos) at (0,0.3em);
    \fill[sectionbg] (current page.north west) rectangle (current page.north east |- sbpos);}%
  \hspace*{-\pagemargin}%
  \colorbox{sectionbg}{%
    \parbox[c][1.9em]{\dimexpr\linewidth+2\pagemargin\relax}{%
      \hspace*{\pagemargin}#1#2}}}}}
\titleformat{\section}[block]
  {\sffamily\Large\bfseries\color{paper}}{}{0pt}
  {\sectionbanner{\thesection\quad}}
\titleformat{name=\section,numberless}[block]
  {\sffamily\Large\bfseries\color{paper}}{}{0pt}
  {\sectionbanner{}}
\titlespacing*{\section}{0pt}{1.2em}{0.8em}

\tcbset{
  mythmstyle/.style={
    breakable, enhanced, boxrule=1pt, sharp corners,
    colback=#1!9!paper, colframe=#1, colbacktitle=#1!85!ink,
    coltitle=paper, coltext=ink,
    fonttitle=\sffamily\bfseries, fontupper=\sffamily,

    before skip=10pt, after skip=10pt,
    separator sign={~},
    description delimiters={\hfill\normalfont\sffamily\bfseries}{},
  }
}

% Soft variant (lemma, corollary): tinted title bar with coloured text, so theorem and definition stand out
\tcbset{
  softthmstyle/.style={
    mythmstyle=#1,
    colbacktitle=#1!18!paper, coltitle=#1!85!ink,
    colback=#1!5!paper,
  }
}

% Light style for examples and remarks: grey side rule, inline bold label, no title bar
\tcbset{
  lightstyle/.style={
    breakable, enhanced, frame hidden, boxrule=0pt, sharp corners,
    colback=#1!18!paper, coltext=ink,
    borderline west={2.5pt}{0pt}{#1},
    left=9pt, right=6pt, top=3pt, bottom=3pt,
    before skip=7pt, after skip=7pt,
    fontupper=\sffamily\small,
    attach title to upper={\ },
    fonttitle=\sffamily\bfseries\small\color{ink!75!paper},
    separator sign={.},
    description delimiters={(}{)},
  }
}

% Outlined style (no filled title bar): #1 = colour, #2 = line pattern (solid or dashed)
\tcbset{
  outlinestyle/.style 2 args={
    breakable, enhanced, sharp corners, frame hidden, boxrule=0pt,
    colback=paper, coltext=ink, colbacktitle=paper, coltitle=#1,
    fonttitle=\sffamily\bfseries, fontupper=\sffamily,
    toptitle=4pt, bottomtitle=2pt,
    before skip=10pt, after skip=10pt,
    separator sign={~},
    description delimiters={\hfill\normalfont\sffamily\bfseries}{},
    overlay={\draw[draw=#1,line width=1.2pt,#2]
      ([shift={(0.6pt,0.6pt)}]frame.south west) rectangle ([shift={(-0.6pt,-0.6pt)}]frame.north east);},
  }
}

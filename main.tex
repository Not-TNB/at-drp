\section{Preliminaries -- Point Set Topology}

\subsection{Topological Spaces}
\begin{definition}{Basic Notions (Topological Space, Topology, Open, Closed)}{topspace}
A topological space $(X,T)\in \cat{Top}$ is a pair consisting of a set $X$ and a topology $T$ on $X$ i.e. $T\subseteq P(X)$ satisfying: \begin{enumerate}
    \item $\emptyset,X\in T$,
    \item $U\subset T \implies \bigcup U \in T$ (closed under unions),
    \item $U\subset_{\text{finite}}T\implies \bigcap U\in T$ (closed under finite intersection)
\end{enumerate}
The elements of $T$ are called open sets. If $A\subset X,X\setminus A\in T$ then $A$ is closed.
\end{definition}

As usual we say $X$ to mean $(X,T)$ if $T$ is clear from context.

\begin{theorem}{Topology induced by a Metric}{mettop}
    Let $(X,d)\in \cat{Met}$. Then \[
    T\coloneqq \{U\subset X\mid \forall x\in U,\exists \epsilon\in \R_{>0}:U(x,\epsilon)\subset U\}
    \]
    is the metric topology induced by $d$, where $U(x,\epsilon)=\{y\in X\mid d(y,x)<\epsilon\}$ is the open $\epsilon$-ball around $x$.
\end{theorem}

\begin{example}{Standard \& Extremal Topologies}{}
    \begin{itemize}
        \item The topology induced by the Euclidean metric on $\R^n$ is called the \textbf{standard topology}.
        \item Let $X$ be a set. Then there are two extremal topologies on $X$: \begin{enumerate}
            \item $P(X)$; the \textbf{discrete topology}.
            \item $\{\emptyset,X\}$; the \textbf{trivial topology}.
        \end{enumerate}
    \end{itemize}
\end{example}

We note that not every topological space is metrisable. Also, in AT we tend to work with less exotic and more ``nicer'' topological spaces unlike those seen in AG (e.g. the Zariski topology on $\operatorname{Spec} \Z$).

\begin{definition}{Open Neighbourhood}{}
    Let $(X,T)\in\Top$ and $x\in X$. \begin{itemize}
        \item A subset $U\subset X$ is an \textbf{open neighbourhood} of $x$ if $U$ is open and $x\in U$.
        \item $U$ is a neighbourhood of $x$ if there is an open neighbourhood $V\subset X$ of $x$ with $V\subset U$.
    \end{itemize}
\end{definition}

\begin{definition}{Closure, Interior, Boundary}{}
    Let $(X,T)\in\Top$ and $Y\subset X$. \begin{itemize}
        \item The \textbf{interior} of $Y$ is $Y^\circ$, the largest open subset of $X$ contained in $Y$: \[
        Y^\circ=\bigcup\{U\mid U\in T \land U\subset Y\}
        \]
        \item The \textbf{closure} of $Y$ is $\overline{Y}$, the smallest closed subset of $X$ containing $Y$: \[
        \overline{Y}=\bigcap \{A\mid X\setminus A \in T \land Y\subset A\}
        \]
        \item The \textbf{boundary} of $Y$ is $\bdry Y=\overline Y \cap \overline{X\setminus Y}$
    \end{itemize}
\end{definition}

We now consider the elementary topological constructions of subspaces and products.

\begin{definition}{Subspace Topology}{}
    Let $(X,T)\in\Top$ and $Y\subset X$. Then \[
    \{U\cap Y\mid U\in T\}
    \]
    is the \textbf{subspace} topology on $Y$. If $T$ on $X$ is induced by a metric $d$ then the subspace topology on $Y$ is induced by the restriction $\restr{d}{Y}$.
\end{definition}

\begin{definition}{Product Topology}{}
    Let $(X,T_X),(Y,T_Y)\in\Top$. Then \[
    \{U\subset X\times Y\mid \forall_{(x,y)\in U} \ \exists_{U_X\in T_X} \ \exists_{U_Y\in T_Y}:(x,y)\in U_X\times U_Y\subset U\}
    \]
    is the \textbf{product} topology on $X\times Y$. This satisfies the universal property of the product in $\Top$.\\

    More generally let $(X_i,T_i)_{i\in I}\in \Top$. The product topology on $X=\prod_{i\in I}X_i$ is the coarsest topology which makes the canonical projections $(X\to X_i)_{i\in I}$ continuous.
    \begin{itemize}
        \item A subset $U\subset X$ is open iff. \[\forall_{x\in U} \ \exists_{J\subset_{\text{finite}}I} \ \exists_{(U_j\subset_{\text{op}}X_j)_{j\in J}}:x\in \prod_{j\in J}U_j \times \prod_{i\in I\setminus J}X_i\subset U\]
    \end{itemize}
\end{definition}

\subsection{Continuous Maps}

\begin{definition}{Continuous Maps}{}
    Let $(X,T_X),(Y,T_Y)\in\Top$. A map $f:X\to Y$ is continuous if \[
    \forall U\in T_Y:f\inv(U)\in T_X
    \]
    i.e. if preimages of open sets are also open. These are precisely the morphisms in $\Top$.
\end{definition}

In metric spaces, continuity wrt. the topology induced by the metric coincides with the usual $\epsilon$-$\delta$ notion.

\begin{example}{Continuous Maps}{}
    \begin{itemize}
        \item If $X\in\Set$ and $T,T'$ are topologies on $X$, then the identity $\id_X:X\to X$ is continuous as a map $(X,T)\to (X,T')$ iff. $T'\subset T$ i.e. $T'$ is coarser than $T$.
        \item The arithmetic operators $+,\cdot,-:\R^2\to \R$ and $/:\R\times (\R\setminus\{0\})\to \R$ are continuous wrt. the standard topology.
        \item If $(X,T)\in\Top$ and $Y\subset X$ then $Y\xhookrightarrow{}X$ is continuous wrt. the subspace topology on $Y$. 
    \end{itemize}
\end{example}

\begin{theorem}{Inheritance Properties of Continuous Maps}{}
    Let $(X,T_X),(Y,T_Y),(Z,T_Z)\in\Top$ and $(f:X\to Y),(g:Y\to Z)$ be maps. Then \begin{itemize}
        \item If $f,g$ are continuous then $g\circ f:X\to Z$ is continuous.
        \item If $f$ is continuous and $A\subset X$, then $\restr{f}{A}:A\to Y$ is continuous wrt. the subspace topology on $A$.
        \item The map $f:X\to Y$ is continuous iff. $f:X\to f(X)$ is continuous wrt. the subspace topology on $f(X)$.
    \end{itemize}
\end{theorem}

\begin{theorem}{Glueing Principle for Continuous Maps}{}
    Let $(X,T_X),(Y,T_Y)\in\Top$, let $A,B\subset_{\text{cl}} X$ with $A\cup B=X$ and let $(f:A\to Y),(g:B\to Y)$ be continuous maps (wrt. the subspace topologies) with $\restr{f}{A\cap B}=\restr{g}{A\cap B}$. Then the following is well-defined and continuous. \[
    f\cup_{A\cap B}g:X\to Y \qquad x\mapsto \begin{cases}
        f(x) & x\in A\\g(x) & x\in B
    \end{cases}
    \]
\end{theorem}

\begin{definition}{Homeomorphisms}{}
    Let $(X,T_X),(Y,T_Y)\in\Top$. A continuous map $f:X\to Y$ is a homeomorphism if there is another continuous map $g:Y\to X$ such that \[g\circ f=\id_X\quad\land\quad f\circ g=\id_Y\] If one exists then $X$ and $Y$ are homeomorphic; $X\iso_{\Top}Y$ (these are the isomorphisms in $\Top$).
\end{definition}
Intuitively this corresponds to the idea of deforming a shape into another without tearing or glueing. Note that not every bijective continuous map is a homeomorphism.

\subsection{Path-Connectedness}
Continuous functions $[0,1]\to\R$ satisfy the IVT. We can describe a similar phenomenon in general topological spaces.
\begin{definition}{Path \& Path-Connectedness}{}
Let $(X,T)\in\Top$.
    \begin{itemize}
        \item A \textbf{path} in $X$ is a continuous map $\gamma:[0,1]\to X$ (wrt. standard topology on $[0,1]\subset\R$).
        \item We call $\gamma(0)$ and $\gamma(1)$ the start and end points of $\gamma$. If these coincide then $\gamma$ is closed.
        \item $X$ is path-connected if $\forall_{x,y\in X} \ \exists_{\text{path }\gamma:[0,1]\to X} :\gamma(0)=x\land \gamma(1)=y$
    \end{itemize}
\end{definition}

\begin{example}{}{}
    \begin{itemize}
        \item The unit interval $[0,1]$ is path-connected.
        \item $\forall n\in\N$, the space $\R^n$ is path-connected wrt. the standard topology.
        \item If $X\in\Set$ with $|X|\geq 2$ then $X$ is not path-connected wrt. the discrete topology.
    \end{itemize}
\end{example}

\begin{theorem}{Continuity Preserved Path-connectedness}{}
    Let $(X,T_X),(Y,T_Y)\in\Top$.\begin{itemize}
        \item Let $f:X\to Y$ be continuous. If $X$ is path-connected then $f(X)$ is path-connected.
        \item Path-connectedness is a \textbf{homeomorphism invariant} i.e. if $X\iso_{\Top} Y$ then $X$ is path-connected iff. $Y$ is.
    \end{itemize}
\end{theorem}

A weaker version of path-connectedness is (general) connectedness -- if the only way to partition $X$ into open sets is in the trivial manner.

\begin{definition}{Connectedness}{}
    $(X,T_X)\in\Top$ is connected if $\forall U,V\in T_X:U\cup V=X\land U\cap V=\emptyset \implies U=\emptyset \lor V=\emptyset$
\end{definition}

\begin{theorem}{Path-connectedness $\implies$ Connectedness}{}
    If $(X,T_X)\in\Top$ is path-connected then it is also connected.
\end{theorem}

If $U\subset_{\text{op}}\R^n$ then $U$ is path-connected iff. it is connected. However in other situations there are spaces that are connected but not path connected, like the Topologist's sine curve: \[
\{(x,\sin x\inv)\mid x\in (0,1]\}\cup\{0\}\times [-1,1]\subset \R^2

\subsection{Hausdorff Spaces}

\begin{definition}{Hausdorff Space}{}
    $(X,T_X)\in\Top$ is Hausdorff if every two points can be separated by open sets i.e. \[
    \forall_{x\neq y\in X} \ \exists_{U,V\subset_{\text{op}}X}:x\in U,y\in V,U\cap V=\emptyset
    \]
\end{definition}

\begin{theorem}{Metric Spaces are Hausdorff}{}
    Let $(X,d)\in\cat{Met}$. Then the metric topology on $X$ is Hausdorff.
\end{theorem}

\begin{corollary}{}{}
    If $X\in\Set$ with $\# X\geq 2$ then the trivial topology on $X$ is \textbf{NOT} induced by a metric on $X$.
\end{corollary}

\begin{theorem}{Hausdorff is Homeomorphically Invariant}{}
    If $A\iso_{\Top}B$ then $A$ is Hausdorff iff $B$ is.
\end{theorem}

\subsection{Compactness}

\begin{definition}{Compactness}{}
    $(X,T)\in\Top$ is compact if every open cover of $X$ contains a finite subcover. In other words \[
    \forall {(U_i)_{i\in I}} :X=\bigcup_{i\in I}U_i \implies \left[ \exists {J\subset_{\text{finite}}I} : X=\bigcup_{i\in J}U_i \right]
    \]
\end{definition}

\begin{example}{}{}
    Let $X\in\Set$. \begin{itemize}
        \item $X$ is compact wrt. the trivial topology.
        \item $X$ is compact wrt. the discrete topology iff. $X$ is finite.
    \end{itemize}
\end{example}

Intuitively this is understood as a finiteness property of topological spaces.
We see that $[0,1]$ is compact wrt. the standard topology, meaning that every continuous map $[0,1]\to \R$ attains a minimum and maximum. More generally:

\begin{theorem}{Generalised Maximum Principle}{gmp}
    Let $(X,T_X),(Y,T_Y)\in\Top$. \begin{itemize}
        \item Let $f:X\to Y$ be continuous. If $X$ is compact then so is $f(X)$.
        \item Compactness is homeomorphism invariant.
    \end{itemize}
\end{theorem}

The Heine-Borel theorem gives us a simple characterisation of compact sets in $\R^n$. 

\begin{theorem}{Heine-Borel Theorem}{}
    Let $n\in\N$ and $A\subset \R^n$ endowed with the subspace topology of the standard topology on $\R^n$. The following are equivalent:\begin{enumerate}
        \item $A$ is compact.
        \item $A$ is closed and bounded wrt. the Euclidean metric on $\R^n$.
        \item $A$ is sequentially compact wrt. the Euclidean metric (i.e. every sequence in $A$ has a subsequence converging to a limit in $A$). (This is deduced from Bolzano-Weierstrass)
    \end{enumerate}
\end{theorem}

Every compact subspace of a metric space is closed and bounded, but the converse is not true. An example is infinite sets -- these are closed and bounded wrt. the discrete metric but they are obviously not compact. A more general relationship between closedness and compactness yields a useful sufficient homeomorphism criterion:

\begin{theorem}{Closed vs. Compact}{cvc}
    Let $(X,T)\in\Top$ and let $Y\subset X$. \begin{itemize}
        \item If $X$ is compact and $Y\subset_{\text{cl}}X$ then $Y$ is also compact.
        \item If $X$ is Hausdorff and $Y$ is compact then $Y\subset_{\text{cl}}X$.
    \end{itemize}
\end{theorem}

\begin{corollary}{Compact-Hausdorff Trick}{cht}
    Let $(X,T_X)\in\Top$ be compact and $(Y,T_Y)\in\Top$ be Hausdorff, and let $f:X\to Y$ be continuous and bijective. Then $f$ is a homeomorphism.
\end{corollary}
\begin{proof}[\proofof{cor:cht}]
    $f$ is bijective so it admits an inverse in $\Set$, say $g:Y\to X$. It suffices to show that $g$ is continuous, and equivalently it suffices to show that $f$-images of closed subsets of $X$ are also closed in $Y$.\\

    So we take an arbitrary closed subset of $X$ and observe the following: \begin{align*}
        A\subset_{\text{cl}}X &\implies A  \text{ compact} \tag{Theorem \ref{th:cvc}, since $X$ compact}\\
        &\implies f(A) \text{ compact} \tag{Theorem \ref{th:gmp}, since $f$ continuous}\\
        &\implies f(A)\subset_{\text{cl}}Y \tag{Theorem \ref{th:cvc}, since $Y$ Hausdorff}
    \end{align*}
\end{proof}

Compactness becomes fiddly under taking products.

\begin{theorem}{Product of Two Compact Spaces}{}
    Let $(X,T_X),(Y,T_Y)\in\Top$ be compact. Then $X\times Y$ is also compact wrt. the product topology.
\end{theorem}

\begin{theorem}{Tychonoff's Theorem}{}
    Let $(X_i)_{i\in I}$ be a family of compact spaces. Then $\prod_{i\in I}X_i$ is also compact.
\end{theorem}

\begin{remark}{}{}
    Tychonoff's theorem is foundational to general topology, and is equivalent to the AoC (and hence to Zorn's Lemmas and the Well Ordering Theorem).
\end{remark}

\subsection{Quotients \& Disjoint Unions}

Subspaces and products build maps \emph{into} spaces. To \emph{glue} spaces we need the dual constructions, which are characterised by maps \emph{out of} them.

\begin{definition}{Quotient Topology}{quot}
    Let $(X,T_X)\in\Top$, $Y\in\Set$ and $p:X\to Y$ surjective. The \textbf{quotient topology} on $Y$ induced by $p$ is \[
    \{U\subset Y\mid p\inv(U)\in T_X\}.
    \]
    Usually $Y=X/{\sim}$ for an equivalence relation $\sim$ on $X$, and $p:x\mapsto[x]$ is the canonical projection.
\end{definition}

\begin{remark}{Quotient Maps Need Not Be Open}{}
    $p$ is continuous by construction, but it need not send open sets to open sets. For $p:[0,1]\to[0,1]/(0\sim1)$, the set $U=[0,\tfrac12)$ is open in $[0,1]$, but $p\inv(p(U))=[0,\tfrac12)\cup\{1\}$ is not, so $p(U)$ is not open.
\end{remark}

\begin{example}{Collapsing a Subspace}{collapse}
    Let $X\in\Top$ and $A\subset X$. Then $X/A\coloneqq X/{\sim}$, where $a\sim b$ for all $a,b\in A$ (and $x\sim x$ for all $x$), i.e.\ $A$ is crushed to a single point. For instance $[0,1]/\{0,1\}$ is the circle.
\end{example}

\begin{theorem}{Universal Property of Quotients}{upquot}
    Let $(X,T_X)\in\Top$, $p:X\to Y$ surjective with $Y$ carrying the quotient topology, and $Z\in\Top$. A map $g:Y\to Z$ is continuous iff.\ $g\circ p:X\to Z$ is continuous. Equivalently, every continuous $f:X\to Z$ which is constant on the fibres of $p$ (i.e.\ $x\sim x'\implies f(x)=f(x')$) factors uniquely through a continuous $\bar f:Y\to Z$.
    \[\begin{tikzcd}
        X \arrow[r, "p"] \arrow[rd, "f"'] & Y \arrow[d, dashed, "\exists!\,\bar f"] \\
        & Z
    \end{tikzcd}\]
\end{theorem}
\begin{proof}[\proofof{th:upquot}]
    If $g$ is continuous then so is $g\circ p$. Conversely, if $g\circ p$ is continuous and $U\subset_{\text{op}}Z$, then $p\inv(g\inv(U))=(g\circ p)\inv(U)$ is open in $X$, so $g\inv(U)$ is open in $Y$ by definition of the quotient topology.
\end{proof}

\begin{definition}{Disjoint Union Topology}{dunion}
    Let $(X_i)_{i\in I}$ be a family in $\Top$ with canonical inclusions $\iota_j:X_j\to\bigsqcup_{i\in I}X_i$. The \textbf{disjoint union topology} is \[
    \Big\{U\subset\bigsqcup_{i\in I}X_i \ \Big|\ \forall j\in I:\iota_j\inv(U)\subset_{\text{op}}X_j\Big\}.
    \]
    We usually view each $X_j$ as a subset of $\bigsqcup_{i}X_i$.
\end{definition}

\begin{theorem}{Universal Property of Disjoint Unions}{updunion}
    For every $Z\in\Top$ and family of continuous maps $(f_i:X_i\to Z)_{i\in I}$ there is exactly one continuous $f:\bigsqcup_{i}X_i\to Z$ with $f\circ\iota_i=f_i$ for all $i$. That is, $\bigsqcup_i X_i$ is the \textbf{coproduct} in $\Top$.
\end{theorem}

Quotients and disjoint unions are both colimits in $\Top$.

If $X$ is a topological space, then the product $X\times I,I=[0,1]$ allows us to model the ``timeline`` of points in $X$ through the ``time`` interval $I$; this will be useful in defining deformation concepts like homotopy.\\

\begin{remark}{Universal Property of Products in $\Top$}{upprod}
    Let $X_1,X_2\in\Top$ and let $p_k:X_1\times X_2\to X_k$ be the projections, which are continuous for the product topology. Then for every $Y\in\Top$ and all continuous $f_1:Y\to X_1$, $f_2:Y\to X_2$ there is exactly one continuous $f:Y\to X_1\times X_2$ with $p_1\circ f=f_1$ and $p_2\circ f=f_2$:
    \[\begin{tikzcd}[column sep=large, row sep=large]
        & Y \arrow[ld, "f_1"'] \arrow[d, dashed, "\exists!\,f"] \arrow[rd, "f_2"] & \\
        X_1 & X_1\times X_2 \arrow[l, "p_1"] \arrow[r, "p_2"'] & X_2
    \end{tikzcd}\]
    Specifically we have $f=(f_1,f_2)$. For continuity it suffices to check basic open sets: $f\inv(U_1\times U_2)=f_1\inv(U_1)\cap f_2\inv(U_2)$ is open. Hence $(X_1\times X_2,p_1,p_2)$ is the categorical product in $\Top$, and the product topology is the coarsest topology on $X_1\times X_2$ making both projections continuous. The same holds for arbitrary families $(X_i)_{i\in I}$ with $\prod_{i\in I}X_i$.
\end{remark}

\subsubsection{Construction via Quotients}

Many spaces are describable by \textbf{glueing}, where we begin with a space and declare which sets of points are to be collapsed together. We saw formally that for $X\in\Top$ and an equivalence relation $\sim$ on $X$, we give $X/{\sim}$ the quotient topology (Definition \ref{def:quot}), i.e. the finest topology making the projection $p:X\to X/{\sim}$ continuous.

\begin{example}{Möbius Strip}{mobius}
    The \textbf{Möbius strip} is $M\coloneqq\big([0,1]\times[0,1]\big)/{\sim}$, where $[0,1]^2$ carries the product topology and for $x,y\in[0,1]^2$: \[
    x\sim y\iff x=y\ \lor\ \big(x_1\in\{0,1\}\land y_1=1-x_1\land y_2=1-x_2\big),
    \]
    i.e.\ the two vertical edges are glued with a half twist, $(0,t)\sim(1,1-t)$ (Figure \ref{fig:mobius}). We could also describe $M$ as a subspace of $\R^3$, but the glueing description is simpler and topologically more illuminating.
\end{example}

\begin{example}{Real Projective Space}{rpn}
    Let $n\in\N$. The \textbf{real projective space} is $\RP{n}\coloneqq\Sph{n}/{\sim}$, where $\sim$ is generated by $x\sim -x$ for all $x\in\Sph{n}$; equivalently, the space of lines through $0$ in $\R^{n+1}$. Note $\RP{1}\iso_{\Top}\Sph{1}$, but $\RP{2}$ does not embed in $\R^3$.
\end{example}

\newpage
A special case is collapsing a subspace $A\subset X$ to a point, $X/A$. Another one is collapsing the boundary of an $n$-ball:

\begin{proposition}{Collapsing the Boundary of a Ball}{dnsn}
    For all $n\in\N_{>0}$ we have $\Disk{n}/\Sph{n-1}\iso_{\Top}\Sph{n}$.
\end{proposition}
\begin{proof}[\proofof{th:dnsn}]
    Define $f:\Disk{n}\to\Sph{n}\subset\R^n\times\R$ by \[
    f(x)\coloneqq\Big(\sin(\pi\norm{x})\,\frac{x}{\norm{x}},\ \cos(\pi\norm{x})\Big)\quad (x\neq0),\qquad f(0)\coloneqq(0,1).
    \]
    It is continuous (at $0$: $\norm{\sin(\pi\norm{x})\,x/\norm{x}}\leq\pi\norm{x}\to0$), maps the radius-$r$ sphere onto the latitude at height $\cos(\pi r)$, and sends all of $\Sph{n-1}$ to the south pole $(0,-1)$. Hence $f$ is constant on $\Sph{n-1}$ and injective on $\Disk{n}\setminus\Sph{n-1}$, and surjective. By Theorem \ref{th:upquot} it induces a continuous bijection \[
    \bar f:\Disk{n}/\Sph{n-1}\to\Sph{n}.
    \]
    $\Disk{n}/\Sph{n-1}$ is compact (continuous image of the compact $\Disk{n}$) and $\Sph{n}$ is Hausdorff, so $\bar f$ is a homeomorphism by Corollary \ref{cor:cht}.
\end{proof}

\subsubsection{Construction via Pushouts}

Glueing several spaces along common pieces combines disjoint unions and quotients:

\begin{definition}{Pushout of Topological Spaces}{pushout}
    Let $X_0,X_1,X_2\in\Top$ and $i_1:X_0\to X_1$, $i_2:X_0\to X_2$ continuous. The \textbf{pushout} of
    \[\begin{tikzcd}
        X_0 \arrow[r, "i_2"] \arrow[d, "i_1"'] & X_2 \\
        X_1 &
    \end{tikzcd}\]
    is $X_1\cup_{X_0}X_2\coloneqq(X_1\sqcup X_2)/{\sim}$, with $\sim$ generated by $i_1(x)\sim i_2(x)$ for all $x\in X_0$, carrying the quotient topology of the disjoint union topology.
\end{definition}

\newpage

\begin{proposition}{Universal Property of Pushouts in $\cat{Top}$}{uppush}
    Let $j_1:X_1\to X_1\cup_{X_0}X_2$ and $j_2:X_2\to X_1\cup_{X_0}X_2$ be induced by the inclusions into $X_1\sqcup X_2$. Then $j_1\circ i_1=j_2\circ i_2$, and for all $Z\in\Top$ and continuous $f_1:X_1\to Z$, $f_2:X_2\to Z$ with $f_1\circ i_1=f_2\circ i_2$ there is a unique continuous $f:X_1\cup_{X_0}X_2\to Z$ with $f\circ j_1=f_1$ and $f\circ j_2=f_2$.
    \[\begin{tikzcd}
        X_0 \arrow[r, "i_2"] \arrow[d, "i_1"'] & X_2 \arrow[d, "j_2"] \arrow[ddr, bend left, "f_2"] & \\
        X_1 \arrow[r, "j_1"'] \arrow[drr, bend right, "f_1"'] & X_1\cup_{X_0}X_2 \arrow[dr, dashed, "\exists!\,f"] & \\
        & & Z
    \end{tikzcd}\]
    Moreover, any $(X',j_1',j_2')$ with this universal property is homeomorphic to $X_1\cup_{X_0}X_2$ via a unique homeomorphism compatible with the $j$'s.
\end{proposition}
\begin{proof}[\proofof{th:uppush}]
    By Theorem \ref{th:updunion}, $f_1,f_2$ induce a unique continuous $g:X_1\sqcup X_2\to Z$. Since $f_1\circ i_1=f_2\circ i_2$, $g$ is constant on the generating relations $i_1(x)\sim i_2(x)$, hence on equivalence classes, so it descends uniquely to a continuous $f$ on the quotient by Theorem \ref{th:upquot}. Uniqueness of $X'$ -- the induced maps $X\to X'$ and $X'\to X$ compose to maps satisfying the property of $\id$, so by uniqueness they are exactly $\id$.
\end{proof}

\subsubsection{The Homeomorphism Problem}

Although the problem of classifying homeomorphic spaces is undecidable, there remain strategies to prove that two given spaces are homeomorphic (or not). We could follow the definition of homeomorphism, explicitly constructing continuous maps $X\to Y,Y\to X$ which are mutual inverses. However, in the often case that $X$ and $Y$ are universal constructions we can typically obtain such a continuous map in one of these directions. It then remains to show that this map is a homeomorphism -- the compact-Hausdorff trick (Corollary \ref{cor:cht}) proves useful.

\begin{example}{The Circle}{}
    Let $C\coloneq [0,1]/(0\sim 1)$ (endowed with the quotient topology of the standard topology on $[0,1]$). We prove that $C\iso_{\Top}S^1$: \begin{itemize}
        \item $C$ is a quotient space so by the universal property of quotient spaces we construct \[
        f:[0,1]\to S^1 \qquad t\mapsto (\cos(2\pi t),\sin(2\pi t))
        \]
        and this is a well-defined continuous map. Also $f(0)=f(1)$ so $f$ induces a well-defined map $\overline{f}:C\to S_1$, and because $f$ is continuous then so is $\overline{f}$ by the universal property.
        \item $\overline{f}$ is obviously bijective.
        \item $C$ is compact as a continuous image of the compact $[0,1]$, and $S^1$ is Hausdorff as a subspace of the Hausdorff $\R^2$.
        \item Thus by Corollary \ref{cor:cht}, $\overline{f}$ is a homeomorphism.
    \end{itemize}
\end{example}

We may also want to know \textbf{when two spaces are not homeomorphic}. Since it is obviously infeasible to determine and check all continuous maps $X\to Y,Y\to X$, we instead are interested in considering \textbf{homeomorphism invariants} which have different values in $X$ and $Y$ to form a contradiction.

Basic point-set properties are the first examples: \textbf{path-connectedness}, \textbf{connectedness}, \textbf{Hausdorffness} and \textbf{compactness} are all homeomorphism invariant. For instance, $[0,1]\not\iso_{\Top}\R$ (compactness), and $\R\not\iso_{\Top}\R^2$, since removing a point disconnects $\R$ but not $\R^2$. However these invariants are not sufficient for a numerous number of examples. Algeraic topology instead attaches \emph{algebraic} invariants (groups, homomorphisms) to spaces and maps.

\subsubsection{The Categories of Algebraic Topology}

\begin{example}{Topological Categories}{topcats}
    \begin{itemize}
        \item $\Top$: topological spaces and continuous maps.
        \item $\Toppt$: \textbf{pointed spaces} $(X,x_0)$ with $x_0\in X$; morphisms $(X,x_0)\to(Y,y_0)$ are continuous $f:X\to Y$ with $f(x_0)=y_0$. This is the home of $\pi_1$.
        \item $\Toppair$: \textbf{pairs of spaces} $(X,A)$ with $A\subset X$; morphisms $(X,A)\to(Y,B)$ are continuous $f:X\to Y$ with $f(A)\subset B$. This is the home of relative homology. We regard $\Top\subset\Toppair$ via $X\mapsto(X,\emptyset)$.
        \item $\hTop$: spaces and \emph{homotopy classes} of maps $[X,Y]$.
    \end{itemize}
\end{example}

\begin{definition}{Simplex Category}{simpcat}
    The \textbf{simplex category} $\Delta$ has objects $[n]\coloneqq\{0,1,\dots,n\}$ for $n\in\N$, and morphisms $[n]\to[m]$ the monotone (order-preserving) maps. Distinguished morphisms are the
    \begin{itemize}
        \item \textbf{coface maps} $\delta^i:[n-1]\to[n]$, the injection skipping $i$;
        \item \textbf{codegeneracy maps} $\sigma^i:[n+1]\to[n]$, the surjection hitting $i$ twice.
    \end{itemize}
    Every morphism in $\Delta$ is a composite of these.
\end{definition}

\begin{example}{Algebraic Categories}{algcats}
    $\Set$, $\Grp$, $\Ab$, $\Mod_R$, $\Vect_K$, and $\Ch$: chain complexes $(C_\bullet,\bdry)$ of abelian groups with $\bdry\circ\bdry=0$, and chain maps $f_\bullet$ with $\bdry\circ f=f\circ\bdry$. Homology will be a functor $H_n:\Ch\to\Ab$.
\end{example}

\subsubsection{Functors as Invariants}

\begin{proposition}{Functors Preserve Isomorphisms}{funciso}
    Let $F:\cat{C}\to\cat{D}$ be a functor and $f:X\to Y$ an isomorphism in $\cat{C}$. Then $F(f)$ is an isomorphism in $\cat{D}$. In particular, $X\iso_{\cat{C}}Y\implies F(X)\iso_{\cat{D}}F(Y)$.
\end{proposition}
\begin{proof}[\proofof{th:funciso}]
    If $g\circ f=\id_X$ and $f\circ g=\id_Y$, then $F(g)\circ F(f)=F(g\circ f)=F(\id_X)=\id_{F(X)}$, and symmetrically.
\end{proof}

\begin{corollary}{Functorial Invariants}{funcinv}
    Let $F:\Top\to\cat{D}$ be a functor. If $F(X)\not\iso_{\cat{D}}F(Y)$, then $X\not\iso_{\Top}Y$.
\end{corollary}

\begin{remark}{Converse Fails}{}
    $F(X)\iso F(Y)$ says nothing about $X$ and $Y$: the forgetful functor $U:\Top\to\Set$ gives $U(\R)\iso_{\Set}U(\R^2)$, yet $\R\not\iso_{\Top}\R^2$ (below). A good invariant is fine enough to distinguish the spaces we care about, yet computable.
\end{remark}

\begin{example}{Basic Functors}{basicfun}
    \begin{itemize}
        \item The \textbf{forgetful functor} $U:\Top\to\Set$; it is represented by the point, $U\iso\Top(\pt,-)$.
        \item The \textbf{standard simplices} assemble to a functor $\Delta^\bullet:\Delta\to\Top$, $[n]\mapsto\Simp{n}$, sending monotone $f:[n]\to[m]$ to the affine map \[
        \Simp{n}\to\Simp{m},\qquad \textstyle\sum_{i}t_ie_i\mapsto\sum_i t_ie_{f(i)}.
        \]
        In particular $\delta^i$ gives the inclusion of the $i$-th face $\Simp{n-1}\inj\Simp{n}$.
    \end{itemize}
\end{example}

\begin{definition}{Path Components}{pi0}
    For $X\in\Top$, write $x\sim y$ if there is a path from $x$ to $y$ (an equivalence relation: constant, reversed and concatenated paths). The set of \textbf{path components} is $\pi_0(X)\coloneqq X/{\sim}$. A continuous $f:X\to Y$ induces $\pi_0(f):[x]\mapsto[f(x)]$, well defined since $f\circ\gamma$ is a path. This gives a functor $\pi_0:\Top\to\Set$; $X$ is path-connected iff.\ $\#\pi_0(X)=1$.
\end{definition}

\begin{proposition}{$\R$ is Not Homeomorphic to $\R^n$}{rrn}
    For all $n\geq2$ we have $\R\not\iso_{\Top}\R^n$.
\end{proposition}
\begin{proof}[\proofof{th:rrn}]
    Suppose $h:\R\to\R^n$ is a homeomorphism. Then $h$ restricts to a homeomorphism $\R\setminus\{0\}\to\R^n\setminus\{h(0)\}$ (with inverse the restriction of $h\inv$; restrictions of continuous maps are continuous). By Proposition \ref{th:funciso}, $\pi_0$ yields a bijection between
    \begin{itemize}
        \item $\pi_0(\R\setminus\{0\})$, which has \textbf{two} elements: $(-\infty,0)$ and $(0,\infty)$ are path-connected, and no path joins $-1$ to $1$ in $\R\setminus\{0\}$ by the IVT;
        \item $\pi_0(\R^n\setminus\{p\})$, which has \textbf{one} element: any two points can be joined by a straight line, or, if that line passes through $p$, by two straight lines via a point off it (possible as $n\geq2$).
    \end{itemize}
    Contradiction.
\end{proof}

\end{remark}
